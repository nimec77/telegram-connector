```rust
/// Telegram public-username shape: 5–32 chars of `[A-Za-z0-9_]`, not
/// starting with a digit. Used to reject malformed usernames locally before
/// spending a resolve RPC (work-order D8). Deliberately stricter than
/// [`Username::new`], which also admits Telegram-supplied Unicode names
/// arriving from the API side.
pub fn is_valid_shape(s: &str) -> bool {
    (5..=32).contains(&s.len())
        && s.chars().all(|c| c.is_ascii_alphanumeric() || c == '_')
        && !s.starts_with(|c: char| c.is_ascii_digit())
}
```

```rust
    /// Try to acquire tokens; on failure returns (available, retry_after_seconds)
    fn try_acquire(&mut self, tokens: u32) -> Result<(), (f64, u64)> {
        self.refill();

        let tokens_f64 = tokens as f64;
        if self.available_tokens >= tokens_f64 {
            self.available_tokens -= tokens_f64;
            Ok(())
        } else {
            let tokens_needed = tokens_f64 - self.available_tokens;
            let retry_after = (tokens_needed / self.refill_rate).ceil() as u64;
            Err((self.available_tokens, retry_after))
        }
    }
```

```rust
/// Replace base64 image content blocks with size-annotated stubs (work-order
/// D6): the replay tool exists for when context is already damaged, so binary
/// payloads only replay on explicit request.
fn omit_binary_blocks(response: &mut serde_json::Value) {
    let Some(blocks) = response
        .get_mut("result")
        .and_then(|r| r.get_mut("content"))
        .and_then(|c| c.as_array_mut())
    else {
        return;
    };
    for block in blocks {
        if block.get("type").and_then(|t| t.as_str()) != Some("image") {
            continue;
        }
        let Some(data) = block.get("data").and_then(|d| d.as_str()) else {
            continue;
        };
        let padding = data.bytes().rev().take_while(|&b| b == b'=').count();
        let size_bytes = data.len() / 4 * 3 - padding;
        let mime_type = block.get("mimeType").cloned().unwrap_or(serde_json::Value::Null);
        *block = serde_json::json!({
            "type": "image",
            "omitted": true,
            "mime_type": mime_type,
            "size_bytes": size_bytes,
        });
    }
}
```
